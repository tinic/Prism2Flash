cmake_minimum_required(VERSION 3.20)
project(Prism2Flash VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(PRISM2_AMIGA "Build the native Amiga PCMCIA backend" OFF)
add_library(prism2_core src/prism2.c)
target_include_directories(prism2_core PUBLIC include)
target_compile_options(prism2_core PRIVATE -Wall -Wextra -Werror)
if(PRISM2_AMIGA)
  add_executable(Prism2Genesis src/main.c src/amiga.c)
  target_compile_definitions(Prism2Genesis PRIVATE PRISM2_AMIGA=1)
  target_include_directories(Prism2Genesis PRIVATE third_party/pccard/include)
  target_link_libraries(Prism2Genesis PRIVATE prism2_core)
  target_compile_options(Prism2Genesis PRIVATE -Wall -Wextra -Werror)
  install(TARGETS Prism2Genesis RUNTIME DESTINATION C)
else()
  add_library(prism2_mock tests/mock.c)
  target_include_directories(prism2_mock PUBLIC tests)
  target_link_libraries(prism2_mock PUBLIC prism2_core)
  target_compile_options(prism2_mock PRIVATE -Wall -Wextra -Werror)
  add_executable(Prism2GenesisSim src/main.c tests/sim_platform.c)
  target_link_libraries(Prism2GenesisSim PRIVATE prism2_mock)
  target_compile_options(Prism2GenesisSim PRIVATE -Wall -Wextra -Werror)
  enable_testing()
  add_executable(test_genesis tests/test_genesis.c)
  target_link_libraries(test_genesis PRIVATE prism2_mock)
  target_compile_options(test_genesis PRIVATE -Wall -Wextra -Werror)
  add_test(NAME genesis_protocol COMMAND test_genesis)
  add_test(NAME cli_info COMMAND Prism2GenesisSim info)
  add_test(NAME cli_probe_x8 COMMAND Prism2GenesisSim probe --sram 8)
  add_test(NAME cli_probe_x16 COMMAND Prism2GenesisSim probe --sram 16)
  add_test(NAME cli_reject_missing_sram COMMAND Prism2GenesisSim probe)
  set_tests_properties(cli_reject_missing_sram PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli_reject_flash COMMAND Prism2GenesisSim flash)
  set_tests_properties(cli_reject_flash PROPERTIES WILL_FAIL TRUE)
endif()
