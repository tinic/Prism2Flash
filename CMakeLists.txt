cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" P2_VERSION LIMIT_COUNT 1)
if(NOT P2_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-[0-9A-Za-z.-]+)?$")
  message(FATAL_ERROR "VERSION must contain a semantic version")
endif()
set(P2_VERSION_BASE "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
project(Prism2Flash VERSION "${P2_VERSION_BASE}" LANGUAGES C)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/RELEASE_DATE" P2_RELEASE_DATE LIMIT_COUNT 1)
if(NOT P2_RELEASE_DATE MATCHES "^([0-9][0-9][0-9][0-9])-([0-9][0-9])-([0-9][0-9])$")
  message(FATAL_ERROR "RELEASE_DATE must contain YYYY-MM-DD")
endif()
set(P2_AMIGA_DATE "${CMAKE_MATCH_3}.${CMAKE_MATCH_2}.${CMAKE_MATCH_1}")
configure_file(include/version.h.in generated/version.h @ONLY)
configure_file(cmake/build-config.json.in generated/build-config.json @ONLY)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(PRISM2_AMIGA "Build the native Amiga PCMCIA backend" OFF)
add_library(prism2_data src/pda.c)
target_include_directories(prism2_data PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(prism2_data PRIVATE -Wall -Wextra -Werror)
add_library(prism2_core src/prism2.c)
target_link_libraries(prism2_core PUBLIC prism2_data)
target_include_directories(prism2_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(prism2_core PRIVATE -Wall -Wextra -Werror)
add_library(prism2_backup src/backup.c src/sha256.c)
target_link_libraries(prism2_backup PUBLIC prism2_core)
target_compile_options(prism2_backup PRIVATE -Wall -Wextra -Werror)
add_library(prism2_firmware src/firmware.c src/fw_validate.c)
target_link_libraries(prism2_firmware PUBLIC prism2_data)
target_compile_options(prism2_firmware PRIVATE -Wall -Wextra -Werror)
add_executable(Prism2Firmware src/firmware_main.c)
target_link_libraries(Prism2Firmware PRIVATE prism2_firmware)
target_compile_options(Prism2Firmware PRIVATE -Wall -Wextra -Werror)
if(PRISM2_AMIGA)
  target_compile_definitions(prism2_backup PRIVATE PRISM2_AMIGA=1)
  target_compile_definitions(Prism2Firmware PRIVATE PRISM2_AMIGA=1)
  add_executable(Prism2Genesis src/main.c src/amiga.c)
  target_compile_definitions(Prism2Genesis PRIVATE PRISM2_AMIGA=1)
  target_include_directories(Prism2Genesis PRIVATE third_party/pccard/include)
  target_link_libraries(Prism2Genesis PRIVATE prism2_backup)
  target_compile_options(Prism2Genesis PRIVATE -Wall -Wextra -Werror)
  add_custom_command(TARGET Prism2Genesis POST_BUILD
    COMMAND "${CMAKE_COMMAND}" "-DBINARY=$<TARGET_FILE:Prism2Genesis>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/check-no-float.cmake")
  add_custom_command(TARGET Prism2Firmware POST_BUILD
    COMMAND "${CMAKE_COMMAND}" "-DBINARY=$<TARGET_FILE:Prism2Firmware>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/check-no-float.cmake")
  add_executable(emulator_env tests/emulator_env.c src/sha256.c)
  target_include_directories(emulator_env PRIVATE include)
  target_compile_options(emulator_env PRIVATE -Wall -Wextra -Werror)
  install(TARGETS Prism2Genesis Prism2Firmware RUNTIME DESTINATION C)
else()
  add_library(prism2_mock tests/mock.c)
  target_include_directories(prism2_mock PUBLIC tests)
  target_link_libraries(prism2_mock PUBLIC prism2_core)
  target_compile_options(prism2_mock PRIVATE -Wall -Wextra -Werror)
  add_executable(Prism2GenesisSim src/main.c tests/sim_platform.c)
  target_link_libraries(Prism2GenesisSim PRIVATE prism2_backup prism2_mock)
  target_compile_options(Prism2GenesisSim PRIVATE -Wall -Wextra -Werror)
  enable_testing()
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  add_executable(test_firmware_api tests/test_firmware_api.c)
  target_link_libraries(test_firmware_api PRIVATE prism2_firmware)
  target_compile_options(test_firmware_api PRIVATE -Wall -Wextra -Werror)
  add_test(NAME firmware_api COMMAND test_firmware_api)
  add_test(NAME firmware_offline COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_firmware.py")
  set_tests_properties(firmware_offline PROPERTIES ENVIRONMENT "P2_FIRMWARE_BIN=$<TARGET_FILE:Prism2Firmware>")
  add_test(NAME release_format COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_release.py")
  add_executable(test_genesis tests/test_genesis.c)
  target_link_libraries(test_genesis PRIVATE prism2_mock)
  target_compile_options(test_genesis PRIVATE -Wall -Wextra -Werror)
  add_test(NAME genesis_protocol COMMAND test_genesis)
  # Exercise the native refusal path with a backend that aborts on any call.
  add_executable(test_backup_gate src/main.c tests/blocked_platform.c)
  target_compile_definitions(test_backup_gate PRIVATE PRISM2_REQUIRE_BACKUP=1)
  target_link_libraries(test_backup_gate PRIVATE prism2_backup)
  target_compile_options(test_backup_gate PRIVATE -Wall -Wextra -Werror)
  add_test(NAME backup_required COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_backup_gate.py" "$<TARGET_FILE:test_backup_gate>")
  add_executable(test_nv_backup tests/test_nv_backup.c tests/sim_platform.c)
  target_link_libraries(test_nv_backup PRIVATE prism2_backup prism2_mock)
  target_compile_options(test_nv_backup PRIVATE -Wall -Wextra -Werror)
  add_test(NAME nv_backup_protocol COMMAND test_nv_backup)
  add_test(NAME nv_backup_cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_nv_backup.py" "$<TARGET_FILE:Prism2GenesisSim>")
  add_test(NAME cli_info COMMAND Prism2GenesisSim info)
  add_test(NAME cli_version COMMAND Prism2GenesisSim --version)
  set_tests_properties(cli_version PROPERTIES PASS_REGULAR_EXPRESSION "Prism2Genesis ${P2_VERSION}")
  add_test(NAME cli_probe_x8 COMMAND Prism2GenesisSim probe --sram 8)
  add_test(NAME cli_probe_x16 COMMAND Prism2GenesisSim probe --sram 16)
  add_test(NAME cli_reject_missing_sram COMMAND Prism2GenesisSim probe)
  set_tests_properties(cli_reject_missing_sram PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli_reject_flash COMMAND Prism2GenesisSim flash)
  set_tests_properties(cli_reject_flash PROPERTIES WILL_FAIL TRUE)
endif()
